using BepInEx.Configuration;
using UnityEngine;

namespace TK2.Customization;

// New recipe compiles into the same plugin. The host limits gameplay recipes to
// the supported local/offline build. This example uses editable recovered logic.
public sealed class RecipeName : IModRecipe
{
    public string Name => "RecipeName";
    public bool ChangesGameplay => true;
    private ConfigEntry<float> _hop = null!;
    private ConfigEntry<KeyCode> _key = null!;

    public void Configure(ConfigFile config)
    {
        _hop = config.Bind("Recipe." + Name, "UpwardVelocity", 6f,
            new ConfigDescription("Extra upward velocity when the shortcut is pressed.", new AcceptableValueRange<float>(0, 30)));
        _key = config.Bind("Recipe." + Name, "Shortcut", KeyCode.F9, "Keyboard shortcut for this example kart hop.");
    }

    public void Tick()
    {
        if (Time.timeScale <= 0 || Ant_CurrentGameConfiguration.eCurrentRaceState !=
            Ant_CurrentGameConfiguration.ERaceState.E_RACE_RUNNING || !Input.GetKeyDown(_key.Value)) return;
        foreach (var kart in UnityEngine.Object.FindObjectsOfType<PixelKartPhysics>())
            if (kart.kartController?.parentPlayer?.ePlayerType == Ant_Player.EPlayerType.E_HUMAN_LOCAL)
                ReadableGame.AddVelocity(kart, Vector3.up * _hop.Value);
    }

    public void Restore() { } // An impulse ends naturally; no persistent setting is replaced.
}
